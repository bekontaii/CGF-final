cmake_minimum_required(VERSION 3.21)
project(WeatherWorld VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(WEATHER_BUILD_TESTS "Build offline logic tests" ON)
option(GLFW_BUILD_DOCS "" OFF)
option(GLFW_BUILD_EXAMPLES "" OFF)
option(GLFW_BUILD_TESTS "" OFF)
option(GLFW_INSTALL "" OFF)
if(UNIX AND NOT APPLE)
  set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "Use documented X11 desktop path")
endif()
add_subdirectory(vendor/glfw EXCLUDE_FROM_ALL)
find_package(OpenGL REQUIRED)
find_package(Threads REQUIRED)
if(NOT WIN32)
  find_package(CURL REQUIRED)
endif()
add_library(glad STATIC vendor/glad/src/glad.c)
target_include_directories(glad PUBLIC vendor/glad/include)
add_library(imgui STATIC
  vendor/imgui/imgui.cpp vendor/imgui/imgui_draw.cpp
  vendor/imgui/imgui_tables.cpp vendor/imgui/imgui_widgets.cpp
  vendor/imgui/backends/imgui_impl_glfw.cpp vendor/imgui/backends/imgui_impl_opengl3.cpp)
target_include_directories(imgui PUBLIC vendor/imgui vendor/imgui/backends)
target_link_libraries(imgui PUBLIC glfw OpenGL::GL)
target_compile_definitions(imgui PUBLIC GLFW_INCLUDE_NONE)
add_library(weather_logic STATIC src/weather.cpp)
target_include_directories(weather_logic PUBLIC include vendor/json/include)
target_link_libraries(weather_logic PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(weather_logic PUBLIC winhttp)
  target_compile_definitions(weather_logic PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_link_libraries(weather_logic PUBLIC CURL::libcurl)
endif()
add_executable(WeatherWorld src/main.cpp src/renderer.cpp)
target_include_directories(WeatherWorld PRIVATE include vendor/glm vendor/stb)
target_link_libraries(WeatherWorld PRIVATE weather_logic glad imgui OpenGL::GL ${CMAKE_DL_LIBS})
target_compile_definitions(WeatherWorld PRIVATE GLFW_INCLUDE_NONE WEATHER_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_compile_options(WeatherWorld PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(weather_logic PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(WeatherWorld PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(weather_logic PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_custom_target(weather_assets ALL
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/shaders" "$<TARGET_FILE_DIR:WeatherWorld>/shaders"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:WeatherWorld>/assets"
  DEPENDS WeatherWorld
  COMMENT "Copying current shaders and team assets"
  VERBATIM)
if(WEATHER_BUILD_TESTS)
  enable_testing()
  add_executable(weather_tests tests/weather_tests.cpp)
  target_link_libraries(weather_tests PRIVATE weather_logic)
  add_test(NAME weather_logic COMMAND weather_tests)
endif()
install(TARGETS WeatherWorld RUNTIME DESTINATION .)
install(DIRECTORY shaders assets DESTINATION .)
